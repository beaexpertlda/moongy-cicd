Declarative YAML Pipelines

Configure builds in plain text versioned with your git repository

.cidc/deploy.yml YAML 
version: '2.4'
name: Production CI/CD & Data Pipeline

on:
  push:
    branches: [main]

jobs:
  security_scan:
    uses: cidc/actions/sast-scanner@v2
    with:
      fail_on_critical: true

  dataops_validate:
    needs: security_scan
    steps:
      - name: Run Database Dry Schema Migration
        run: cidc-data-cli migrate --dry-run
      - name: Validate dbt Models
        run: dbt test --target prod

  k8s_deploy:
    needs: dataops_validate
    strategy: blue-green
    cluster: k8s-prod-eu-west-1
